CREATE TABLE employee (
    emp_id UUID PRIMARY KEY,
    name TEXT,
    department TEXT,
    age INT);

CREATE TABLE salary (
    emp_id UUID,
    month TEXT,
    amount FLOAT,
    PRIMARY KEY (emp_id, month));

##Primary_Key
##Partition Key only (emp_id decides which node stores the row)
CREATE TABLE employees (
    emp_id UUID PRIMARY KEY,
    name TEXT,
    department TEXT);

##Partition Key + Clustering Key (emp_id = partition, month = sort order)
CREATE TABLE salaryy (
    emp_id UUID,
    month TEXT,
    amount FLOAT,
    PRIMARY KEY (emp_id, month));

##Composite Partition Key (data spread across nodes by dept + year)
CREATE TABLE salary_by_dept (
    department TEXT,
    year INT,
    emp_id UUID,
    amount FLOAT,
    PRIMARY KEY ((department, year), emp_id));
